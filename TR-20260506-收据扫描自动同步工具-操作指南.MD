```swift
import Vision
import SwiftUI

struct ReceiptScannerView: UIViewControllerRepresentable {
    @Binding var scannedReceipt: ScannedReceipt?
    @Binding var isPresented: Bool

    func makeUIViewController(context: Context) -> VNDocumentCameraViewController {
        let controller = VNDocumentCameraViewController()
        controller.delegate = context.coordinator
        return controller
    }

    func updateUIViewController(_ uiViewController: VNDocumentCameraViewController, context: Context) {}

    func makeCoordinator() -> Coordinator { Coordinator(self) }

    class Coordinator: NSObject, VNDocumentCameraViewControllerDelegate {
        let parent: ReceiptScannerView

        init(_ parent: ReceiptScannerView) {
            self.parent = parent
        }

        func documentCameraViewController(_ controller: VNDocumentCameraViewController,
                                          didFinishWith scan: VNDocumentCameraScan) {
            let images = (0..<scan.pageCount).map { scan.imageOfPage(at: $0) }
            processScannedImages(images)
            parent.isPresented = false
        }

        private func processScannedImages(_ images: [UIImage]) {
            guard let firstImage = images.first else { return }
            recognizeText(from: firstImage) { [weak self] result in
                DispatchQueue.main.async {
                    self?.parent.scannedReceipt = result
                }
            }
        }

        private func recognizeText(from image: UIImage, completion: @escaping (ScannedReceipt) -> Void) {
            guard let cgImage = image.cgImage else { return }

            let request = VNRecognizeTextRequest { request, error in
                guard let observations = request.results as? [VNRecognizedTextObservation] else { return }

                let recognizedStrings = observations.compactMap { observation in
                    observation.topCandidates(1).first?.string
                }

                let fullText = recognizedStrings.joined(separator: "\n")
                let parsedReceipt = ReceiptParser.parse(from: fullText, rawImage: image)
                completion(parsedReceipt)
            }

            request.recognitionLevel = .accurate
            request.recognitionLanguages = ["en-US"]
            request.usesLanguageCorrection = true

            let handler = VNImageRequestHandler(cgImage: cgImage, options: [:])
            try? handler.perform([request])
        }
    }
}
```

#### 收据数据解析器

```swift
import Foundation

struct ScannedReceipt: Identifiable, Codable {
    let id: UUID
    var merchantName: String
    var totalAmount: Double?
    var taxAmount: Double?
    var date: Date?
    var category: ReceiptCategory
    var lineItems: [LineItem]
    var rawText: String
    var imageData: Data?
    var isVerified: Bool
    var matchedTransactionId: String?
    var syncStatus: SyncStatus
    var createdAt: Date

    enum ReceiptCategory: String, Codable, CaseIterable {
        case meals = "Meals & Entertainment"
        case travel = "Travel"
        case office = "Office Supplies"
        case fuel = "Fuel & Transportation"
        case software = "Software & Subscriptions"
        case marketing = "Marketing & Advertising"
        case utilities = "Utilities"
        case insurance = "Insurance"
        case professional = "Professional Services"
        case other = "Other"
    }

    enum SyncStatus: String, Codable {
        case pending, synced, failed
    }
}

struct LineItem: Identifiable, Codable {
    let id: UUID
    var name: String
    var quantity: Int?
    var unitPrice: Double?
    var totalPrice: Double?
}

class ReceiptParser {
    static func parse(from text: String, rawImage: UIImage? = nil) -> ScannedReceipt {
        let merchantName = extractMerchantName(from: text)
        let totalAmount = extractTotalAmount(from: text)
        let taxAmount = extractTaxAmount(from: text)
        let date = extractDate(from: text)
        let category = categorize(merchant: merchantName, text: text)

        return ScannedReceipt(
            id: UUID(),
            merchantName: merchantName,
            totalAmount: totalAmount,
            taxAmount: taxAmount,
            date: date,
            category: category,
            lineItems: [],
            rawText: text,
            imageData: rawImage?.jpegData(compressionQuality: 0.8),
            isVerified: false,
            matchedTransactionId: nil,
            syncStatus: .pending,
            createdAt: Date()
        )
    }

    private static func extractMerchantName(from text: String) -> String {
        let lines = text.components(separatedBy: .newlines).filter { !$0.trimmingCharacters(in: .whitespaces).isEmpty }
        guard let firstLine = lines.first else { return "Unknown" }

        let cleanLine = firstLine.trimmingCharacters(in: .whitespacesAndNewlines)
        if cleanLine.count > 50 { return "Unknown" }
        return cleanLine
    }

    private static func extractTotalAmount(from text: String) -> Double? {
        let patterns = [
            "total[:\\s]*\\$?([0-9]+\\.?[0-9]*)",
            "amount[:\\s]*\\$?([0-9]+\\.?[0-9]*)",
            "balance[:\\s]*\\$?([0-9]+\\.?[0-9]*)"
        ]

        for pattern in patterns {
            if let regex = try? NSRegularExpression(pattern: pattern, options: .caseInsensitive),
               let match = regex.matches(in: text, range: NSRange(text.startIndex..., in: text)).last,
               let range = Range(match.range(at: 1), in: text),
               let amount = Double(text[range]) {
                return amount
            }
        }
        return nil
    }

    private static func extractTaxAmount(from text: String) -> Double? {
        let pattern = "tax[:\\s]*\\$?([0-9]+\\.?[0-9]*)"
        if let regex = try? NSRegularExpression(pattern: pattern, options: .caseInsensitive),
           let match = regex.matches(in: text, range: NSRange(text.startIndex..., in: text)).first,
           let range = Range(match.range(at: 1), in: text),
           let amount = Double(text[range]) {
            return amount
        }
        return nil
    }

    private static func extractDate(from text: String) -> Date? {
        let patterns = [
            "(0[1-9]|1[0-2])/(0[1-9]|[12][0-9]|3[01])/(20[0-9]{2})",
            "(20[0-9]{2})-(0[1-9]|1[0-2])-(0[1-9]|[12][0-9]|3[01])",
            "(Jan|Feb|Mar|Apr|May|Jun|Jul|Aug|Sep|Oct|Nov|Dec)[a-z]*\\.?\\s+(0?[1-9]|[12][0-9]|3[01]),?\\s+(20[0-9]{2})"
        ]

        let formatter = DateFormatter()
        formatter.locale = Locale(identifier: "en_US_POSIX")

        for (index, pattern) in patterns.enumerated() {
            if let regex = try? NSRegularExpression(pattern: pattern, options: .caseInsensitive),
               let match = regex.matches(in: text, range: NSRange(text.startIndex..., in: text)).first,
               let range = Range(match.range, in: text) {
                let dateString = String(text[range])
                switch index {
                case 0: formatter.dateFormat = "MM/dd/yyyy"
                case 1: formatter.dateFormat = "yyyy-MM-dd"
                case 2: formatter.dateFormat = "MMMM dd, yyyy"
                default: continue
                }
                if let date = formatter.date(from: dateString) { return date }
            }
        }
        return nil
    }

    private static func categorize(merchant: String, text: String) -> ScannedReceipt.ReceiptCategory {
        let lowerMerchant = merchant.lowercased()
        let lowerText = text.lowercased()

        let categoryKeywords: [ScannedReceipt.ReceiptCategory: [String]] = [
            .meals: ["restaurant", "cafe", "coffee", "pizza", "burger", "diner", "grill", "bakery", "starbucks", "dunkin", "chipotle", "mcdonald"],
            .travel: ["airline", "hotel", "motel", "uber", "lyft", "flight", "airbnb", "marriott", "hilton"],
            .fuel: ["shell", "exxon", "chevron", "bp", "gas", "fuel", "station", "sunoco"],
            .office: ["staples", "office depot", "amazon", "best buy", "paper", "ink", "toner"],
            .software: ["apple", "google", "microsoft", "adobe", "subscription", "saas", "app store"],
            .marketing: ["facebook", "google ads", "instagram", "linkedin", "advertising"],
            .utilities: ["electric", "water", "gas bill", "internet", "phone", "verizon", "at&t"],
            .insurance: ["insurance", "geico", "state farm", "allstate", "progressive"],
            .professional: ["legal", "consulting", "accounting", "cpa", "attorney"]
        ]

        for (category, keywords) in categoryKeywords {
            for keyword in keywords {
                if lowerMerchant.contains(keyword) || lowerText.contains(keyword) {
                    return category
                }
            }
        }
        return .other
    }
}
```

### 2. 智能提醒系统 (核心差异化功能)

```swift
import CoreLocation
import UserNotifications

class SmartReminderManager: NSObject, ObservableObject, CLLocationManagerDelegate {
    private let locationManager = CLLocationManager()
    private let notificationCenter = UNUserNotificationCenter.current()
    private var habitTracker = ReceiptHabitTracker()

    override init() {
        super.init()
        locationManager.delegate = self
        locationManager.desiredAccuracy = kCLLocationAccuracyHundredMeters
    }

    func requestPermissions() {
        locationManager.requestAlwaysAuthorization()
        notificationCenter.requestAuthorization(options: [.alert, .sound, .badge]) { _, _ in }
    }

    func startMonitoring() {
        locationManager.startMonitoringSignificantLocationChanges()
        scheduleHabitBasedReminders()
    }

    func locationManager(_ manager: CLLocationManager, didUpdateLocations locations: [CLLocation]) {
        guard let location = locations.last else { return }
        checkNearbyBusinessLocations(location)
    }

    private func checkNearbyBusinessLocations(_ location: CLLocation) {
        let region = CLCircularRegion(center: location.coordinate, radius: 100, identifier: "business_area")
        region.notifyOnEntry = true

        let content = UNMutableNotificationContent()
        content.title = "📸 Snap Your Receipt!"
        content.body = "You're near a business — don't forget to scan your receipt."
        content.sound = .default
        content.categoryIdentifier = "RECEIPT_REMINDER"

        let trigger = UNLocationTrigger(region: region, repeats: true)
        let request = UNNotificationRequest(identifier: UUID().uuidString, content: content, trigger: trigger)

        notificationCenter.add(request)
    }

    private func scheduleHabitBasedReminders() {
        let patterns = habitTracker.getScanPatterns()

        for pattern in patterns where pattern.frequency >= 3 {
            let content = UNMutableNotificationContent()
            content.title = "📸 Receipt Time!"
            content.body = "You usually have expenses around now. Quick scan?"
            content.sound = .default

            var dateComponents = DateComponents()
            dateComponents.hour = pattern.typicalHour
            dateComponents.minute = pattern.typicalMinute

            let trigger = UNCalendarNotificationTrigger(dateMatching: dateComponents, repeats: true)
            let request = UNNotificationRequest(
                identifier: "habit_\(pattern.id)",
                content: content,
                trigger: trigger
            )
            notificationCenter.add(request)
        }
    }
}

struct ScanPattern: Identifiable {
    let id: UUID
    let typicalHour: Int
    let typicalMinute: Int
    let frequency: Int
    let dayOfWeek: Int?
}

class ReceiptHabitTracker {
    private let defaults = UserDefaults.standard
    private let scanHistoryKey = "scan_history"

    func recordScan() {
        var history = getScanHistory()
        history.append(Date())
        if history.count > 100 { history = Array(history.suffix(100)) }
        defaults.set(history.map { $0.timeIntervalSince1970 }, forKey: scanHistoryKey)
    }

    func getScanPatterns() -> [ScanPattern] {
        let history = getScanHistory()
        let calendar = Calendar.current

        var hourCounts: [Int: Int] = [:]
        for date in history {
            let hour = calendar.component(.hour, from: date)
            hourCounts[hour, default: 0] += 1
        }

        return hourCounts.map { hour, count in
            ScanPattern(
                id: UUID(),
                typicalHour: hour,
                typicalMinute: 0,
                frequency: count,
                dayOfWeek: nil
            )
        }.filter { $0.frequency >= 3 }.sorted { $0.frequency > $1.frequency }
    }

    private func getScanHistory() -> [Date] {
        guard let timestamps = defaults.array(forKey: scanHistoryKey) as? [Double] else { return [] }
        return timestamps.map { Date(timeIntervalSince1970: $0) }
    }
```

```swift
import CloudKit

class SyncEngine: ObservableObject {
    private let container: CKContainer
    private let privateDB: CKDatabase
    private let zoneID = CKRecordZone.ID(zoneName: "ReceiptSnap", ownerName: CKCurrentUserDefaultName)

    @Published var syncStatus: SyncState = .idle
    @Published var lastSyncDate: Date?

    enum SyncState {
        case idle, syncing, success, failed(Error)
    }

    init() {
        container = CKContainer(identifier: "iCloud.com.receiptsnap.app")
        privateDB = container.privateCloudDatabase
    }

    func syncReceipt(_ receipt: ScannedReceipt) async throws {
        syncStatus = .syncing
        defer { syncStatus = .idle }

        let record = CKRecord(recordType: "Receipt", recordID: CKRecord.ID(recordName: receipt.id.uuidString, zoneID: zoneID))

        record["merchantName"] = receipt.merchantName as CKRecordValue
        record["totalAmount"] = receipt.totalAmount as? CKRecordValue
        record["taxAmount"] = receipt.taxAmount as? CKRecordValue
        record["date"] = receipt.date as? CKRecordValue
        record["category"] = receipt.category.rawValue as CKRecordValue
        record["isVerified"] = receipt.isVerified as CKRecordValue
        record["rawText"] = receipt.rawText as CKRecordValue
        record["createdAt"] = receipt.createdAt as CKRecordValue

        if let imageData = receipt.imageData {
            let tempURL = FileManager.default.temporaryDirectory.appendingPathComponent("\(receipt.id.uuidString).jpg")
            try imageData.write(to: tempURL)
            let asset = CKAsset(fileURL: tempURL)
            record["receiptImage"] = asset
        }

        try await privateDB.save(record)
        lastSyncDate = Date()
    }

    func fetchReceipts() async throws -> [ScannedReceipt] {
        let query = CKQuery(recordType: "Receipt", predicate: NSPredicate(value: true))
        query.sortDescriptors = [NSSortDescriptor(key: "createdAt", ascending: false)]

        let (matchResults, _) = try await privateDB.records(matching: query, inZoneWith: zoneID)

        return matchResults.compactMap { _, result in
            guard let record = try? result.get() else { return nil }
            return mapRecordToReceipt(record)
        }
    }

    private func mapRecordToReceipt(_ record: CKRecord) -> ScannedReceipt? {
        guard let merchantName = record["merchantName"] as? String,
              let id = UUID(uuidString: record.recordID.recordName) else { return nil }

        return ScannedReceipt(
            id: id,
            merchantName: merchantName,
            totalAmount: record["totalAmount"] as? Double,
            taxAmount: record["taxAmount"] as? Double,
            date: record["date"] as? Date,
            category: ScannedReceipt.ReceiptCategory(rawValue: record["category"] as? String ?? "Other") ?? .other,
            lineItems: [],
            rawText: record["rawText"] as? String ?? "",
            imageData: nil,
            isVerified: record["isVerified"] as? Bool ?? false,
            matchedTransactionId: nil,
            syncStatus: .synced,
            createdAt: record["createdAt"] as? Date ?? Date()
        )
    }
```

```swift
                            - tax_amount: number or null
                            - date: string (YYYY-MM-DD) or null

                            - line_items: array of {name: string, quantity: number or null, unit_price: number or null, total_price: number or null}
                            - payment_method: string or null

                            Return ONLY valid JSON, no markdown.
                            """
                        ]
                    ]
                ]
            ]
        ]

        request.setValue("application/json", forHTTPHeaderField: "Content-Type")

        let (data, response) = try await URLSession.shared.data(for: request)
        guard let httpResponse = response as? HTTPURLResponse, httpResponse.statusCode == 200 else {
            throw ParseError.apiError
        }

        let apiResponse = try JSONDecoder().decode(ClaudeResponse.self, from: data)
        let jsonText = apiResponse.content.first?.text ?? "{}"
        let jsonData = jsonText.data(using: .utf8) ?? Data()

        let parsed = try JSONDecoder().decode(ClaudeReceiptResult.self, from: jsonData)

        return ScannedReceipt(
            id: UUID(),
            merchantName: parsed.merchant_name,
            totalAmount: parsed.total_amount,
            taxAmount: parsed.tax_amount,
            date: parseDate(parsed.date),
            category: ScannedReceipt.ReceiptCategory(rawValue: parsed.category) ?? .other,
            lineItems: parsed.line_items.map { LineItem(id: UUID(), name: $0.name, quantity: $0.quantity, unitPrice: $0.unit_price, totalPrice: $0.total_price) },
            rawText: jsonText,
            imageData: imageData,
            isVerified: false,
            matchedTransactionId: nil,
            syncStatus: .pending,
            createdAt: Date()
        )
    }

    private func parseDate(_ string: String?) -> Date? {
        guard let string = string else { return nil }
        let formatter = DateFormatter()
        formatter.dateFormat = "yyyy-MM-dd"
        return formatter.date(from: string)
    }

    struct ClaudeResponse: Codable {
        let content: [ContentBlock]
        struct ContentBlock: Codable {
            let text: String?
        }
    }

    struct ClaudeReceiptResult: Codable {
        let merchant_name: String
        let total_amount: Double?
        let tax_amount: Double?
        let date: String?
        let category: String
        let line_items: [ClaudeLineItem]
        let payment_method: String?
    }

    struct ClaudeLineItem: Codable {
        let name: String
        let quantity: Int?
        let unit_price: Double?
        let total_price: Double?
    }

    enum ParseError: Error {
        case apiError
        case invalidResponse
    }
}
```

| 规则 | 说明 |
|------|------|
| **架构模式** | MVVM + Repository Pattern；View → ViewModel → Repository → Service |
| **状态管理** | @Observable + @State + @Binding (iOS 17+)；避免Combine |
| **数据持久化** | SwiftData (主) + CloudKit (同步) + UserDefaults (设置) |

| **命名规范** | Swift API Design Guidelines；变量camelCase，类型PascalCase |
| **最低版本** | iOS 17.0 (SwiftData + @Observable) |
| **无第三方依赖** | 仅使用Apple原生框架 + CloudKit + 可选Google Sheets API |

## 🎨 UI设计规范

### 设计原则

| 原则 | 说明 | 实现方式 |
|------|------|----------|
| **零摩擦扫描** | 3步完成：拍→确认→同步 | 底部大"+"按钮，启动即相机 |
| **渐进式展示** | 不一次性展示所有功能 | 首次使用引导，高级功能按需发现 |
| **即时反馈** | 每步操作都有视觉反馈 | 扫描动画、同步状态、成功提示 |
| **暗色优先** | 暗色模式减少眼部疲劳 | 原生Dark Mode，扫描时自动暗色 |
| **触觉反馈** | 关键操作配合触觉 | 保存成功轻震动，删除中等震动 |

## 🏆 竞争优势总结

| 优势 | 详情 | 竞品对比 |
|------|------|----------|
| **智能提醒** | 基于位置+习惯的双引擎提醒，解决"忘记拍照"第一痛点 | 无竞品有此功能 |
| **本地OCR** | Vision Framework免费本地OCR，零API成本 | 多数竞品依赖云端OCR |
| **3步极简** | 拍→确认→同步，比竞品少2-3步 | Expensify需5+步 |
| **隐私优先** | 本地OCR+本地存储，数据不出设备 | 仅TaxLens声称类似 |
| **超低定价** | $4.99/月，比最便宜竞品低24% | SparkReceipt $6.58/月 |
| **免费可用** | 10次/月免费+智能提醒，足够体验 | 优于多数竞品免费版 |
| **自动同步** | CloudKit+iCloud，无需额外账号 | 多数竞品需注册账号 |
| **Google Sheets** | 一键自动同步，小企业最常用工具 | 仅少数竞品支持 |
